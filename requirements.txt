# DO NOT reinstall torch: keep your existing PyTorch 2.7 / CUDA 12.8 build.
# Z-Image is only in diffusers from source, hence the git+ below.

# CAREFUL: the commit is PINNED on purpose.
# A `git+.../diffusers` with no commit takes the HEAD of the day, which is not a
# bound but a time bomb: diffusers main moved to huggingface-hub>=1.23 while
# transformers<5 requires huggingface-hub<1.0 -> ResolutionImpossible at install
# time, for everyone, overnight. This commit is the one THIS repo was validated
# on (it differs from one fork to the next, see requirements-lock.txt).
# To try a more recent version: replace the hash below, run install.bat again,
# and check that `transformers` still installs.
git+https://github.com/huggingface/diffusers@2c7efb95349296cf6bcce981ea036275a82a94df

# transformers >= 4.51 required: Z-Image Turbo uses a Qwen3 encoder (Qwen3Model),
# absent before 4.51 (it fails with "module transformers has no attribute Qwen3Model").
# The <5 bound: the 5.x line has API changes that can break the diffusers
# source. Stay on 4.51+ (tested OK on 4.57).
transformers>=4.51,<5
accelerate
sentencepiece
spandrel

# Required by diffusers to load LoRA weights (load_lora_weights / set_adapters).
# Without it: "PEFT backend is required for this method" -> LoRA silently skipped.
peft

# Z-Image .gguf checkpoints (Q3..Q8 quantised, they stay quantised in VRAM):
# the GGUF reader diffusers uses (GGUFQuantizationConfig) + a layout guard.
gguf

# The before/after comparer = an internal HTML+JS component (see _make_compare_html in app.py).
# No external dep -> a stable stack.
# The <6 bound: Gradio 6 has a Brotli middleware buggy with h11 ("Too little data for
# declared Content-Length") that spams the console when a response is interrupted.
gradio>=4.44,<6

pillow
numpy

# scipy: required by the `beta` sigma schedule offered in the Schedule dropdown.
# FlowMatchEulerDiscreteScheduler raises ImportError on use_beta_sigmas without it, and
# _apply_sampler then falls back to Euler/sgm_uniform -- so choosing beta silently gave
# something else, with one line in the console. karras and exponential are pure numpy and
# were never affected. requirements-lock.txt already pinned scipy==1.14.0, so an ISOLATED
# install had it and an install.bat one did not; that is the gap this line closes.
scipy

# Optional: only for the persistent server mode (--serve, stage 3).
fastapi
uvicorn
