# Optional extras (local tools that don't need a Z-Image model).
# Install:  .venv/Scripts/python -m pip install -r requirements-extra.txt
rembg                  # Remove BG tab (downloads the u2net model on first use)
ultralytics            # 🖐 Detail hands: used ONCE, in a subprocess, to export the
                       # YOLOv8 hand detector (~6 MB, pulled from Bingsu/adetailer) to
                       # ONNX in cache/. It is NEVER imported inside the app process:
                       # loading the torch YOLO model there corrupts the shared model
                       # weights during offload transfers (renders turn to mosaic/NaN).
                       # Runtime detection uses onnxruntime only. Face detailer does
                       # NOT need it (insightface handles faces).
onnx                   # needed by that one-time ONNX export (subprocess).
pillow-avif-plugin     # read .avif input images (else PIL.UnidentifiedImageError)
trustmark              # AI provenance: invisible watermark (write on save if
                       # provenance_watermark=on + read in PNG Info / --provenance).
                       # CPU only (~0.1s/img). Windows: install with PYTHONUTF8=1
                       # (the sdist setup.py crashes on cp1252 otherwise).
c2pa-python            # AI provenance: read C2PA / Content Credentials manifests
                       # (Firefly, ChatGPT, Gemini outputs...) in PNG Info.
